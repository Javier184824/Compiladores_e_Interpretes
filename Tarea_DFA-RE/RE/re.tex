\chapter*{Exercises solved}

\section*{Exercise 12}
The set of strings over \{\textit{a,b,c}\} in which all the \textit{a}'s precede the \textit{b}'s, which in turn precede the \textit{c}'s. It is possible that there are no \textit{a}'s, \textit{b}'s, or \textit{c}'s.
\newline
$$a^*b^*c^*$$
\section*{Exercise 13}
The same set as Exercise 12 without the null string.
\newline
$$aa^*b^*c^* \mid a^*bb^*c^* \mid a^*b^*cc^*$$
\section*{Exercise 14}
The set of strings of length two or more over \{\textit{a,b}\} in which all the \textit{a}'s precede the \textit{b}'s.
\newline
$$(aaa^*b^*) \mid (aa^*bb^*) \mid (bbb^*)$$
\section*{Exercise 15}
The set of strings over \{\textit{a,b}\} that contain the substring \textit{aa} and the substring \textit{bb}.
\newline
$$(a \mid b)^*aa(a \mid b)^*bb(a \mid b)^* \mid (a \mid b)^*bb(a \mid b)^*aa(a \mid b)^*$$

\newpage

\section*{Exercise 16}
The set of strings over \{\textit{a,b}\} in which the substring \textit{aa} occurs at least twice. \textit{Hint}: Beware of the substring \textit{aaa}.
\newline
$$b^*aab^*aab^*$$
\section*{Exercise 17}
The set of strings over \{\textit{a,b,c}\} that do not contain the substring \textit{aa}.
\newline
$$(b \mid c \mid ab \mid ac)^* \mid (b \mid c \mid ab \mid ac)^*a$$
\section*{Exercise 18}
The set of strings over \{\textit{a,b}\} that do not begin with the substring \textit{aaa}.
\newline
$$(a \mid b \mid c) \mid (a \mid b \mid c)(a \mid b \mid c) \mid (b \mid c)(a \mid b \mid c)^* \mid a(b \mid c)(a \mid b \mid c)^* \mid aa(b \mid c)(a \mid b \mid c)^*$$
\section*{Exercise 19}
The set of strings over \{\textit{a,b}\} that do not contain the substring \textit{aaa}.
\newline
$$(b^* \mid ab \mid aab)^* \mid (b^* \mid ab \mid aab)^*(a \mid aa)$$
\section*{Exercise 20}
The set of strings over \{\textit{a,b}\} that do not contain the substring \textit{aba}.
\newline
$$(b \mid a^*)(bbb^* \mid a^*)^* \mid (b \mid a^*)(bbb^* \mid a^*)^*b$$
\section*{Exercise 21}
The set of strings over \{\textit{a,b}\} in which the substring \textit{aa} occurs extacly once.
\newline
$$(b \mid ab)^*aa(b \mid ba)^*$$
\section*{Exercise 22}
The set of strings over \{\textit{a,b,c}\} that begin with \textit{a}, contain exactly two \textit{b}'s, and end with \textit{cc}.
\newline
$$a(a \mid c)^*b(a \mid c)^*b(a \mid c)^*cc$$
\section*{Exercise 23}
The set of strings over \{\textit{a,b}\} that contain the substring \textit{ab} and the substring \textit{ba}.
\newline
$$(a \mid b)^*(ab(a \mid b)^*ba \mid ba(a \mid b)^*ab \mid aba \mid bab)(a \mid b)^* $$
\section*{Exercise 24}
The set of strings over \{\textit{a,b,c}\} that contain the substrings \textit{aa}, \textit{bb} and \textit{cc}.
\[
\begin{aligned}
(a\mid b\mid c)^*\Bigl(
  & aa(a\mid b\mid c)^*bb(a\mid b\mid c)^*cc \\
  & {}\mid aa(a\mid b\mid c)^*cc(a\mid b\mid c)^*bb \\
  & {}\mid bb(a\mid b\mid c)^*aa(a\mid b\mid c)^*cc \\
  & {}\mid bb(a\mid b\mid c)^*cc(a\mid b\mid c)^*aa \\
  & {}\mid cc(a\mid b\mid c)^*aa(a\mid b\mid c)^*bb \\
  & {}\mid cc(a\mid b\mid c)^*bb(a\mid b\mid c)^*aa
\Bigr)(a\mid b\mid c)^*
\end{aligned}
\]
\section*{Exercise 25}
The set of strings over \{\textit{a,b,c}\} in which every \textit{b} is immediately followed by at least one \textit{c}.
\newline
$$(a \mid c \mid bc)^*$$
\section*{Exercise 26}
The set of strings over \{\textit{a,b,c}\} with length three.
\newline
$$(a \mid b \mid c)(a \mid b \mid c)(a \mid b \mid c)$$
\section*{Exercise 27}
The set of strings over \{\textit{a,b,c}\} with length less than three.
\newline
$$(a \mid b \mid c)  \mid  (a \mid b \mid c)(a \mid b \mid c)  \mid  \varepsilon $$
\section*{Exercise 28}
The set of strings over \{\textit{a,b,c}\} with length greater than three.
\newline
$$(a \mid b \mid c)(a \mid b \mid c)(a \mid b \mid c)(a \mid b \mid c)(a \mid b \mid c)^*$$

\section*{Exercise 29}
The set of strings over \{\textit{a,b}\} in which the number of \textit{a}'s is divisible by three.
\newline
$$(b^*ab^*ab^*a)^*$$

\section*{Exercise 30}
The set of strings over \{\textit{a,b,c}\} in which the total number of \textit{a}'s is divisible by three.
\newline
$$\varepsilon  \mid  a^*(b \mid c)a^*(b \mid c)a^*(b \mid c)a^*$$

\section*{Exercise 31}
The set of strings over \{\textit{a,b}\} in which every \textit{a} is either immediately preceded or immediately followed by \textit{b}, for example, \textit{baab}, \textit{aba}, and \textit{b}.
\newline
$$\varepsilon  \mid  (b \mid ab \mid ba)^*$$

\section*{Exercise 32}
The set of strings of odd length over \{\textit{a,b}\} that contain the substring \textit{bb}.
\newline
$$((a \mid b)(a \mid b))^*bb((a \mid b)(a \mid b))^*(a \mid b)  \mid  ((a \mid b)(a \mid b))^*(a \mid b)bb((a \mid b)(a \mid b))^*$$

\section*{Exercise 33}
The set of strings of even length over \{\textit{a,b,c}\} that contain exactly one \textit{a}.
\newline
$$((b \mid c)(b \mid c))^*a(((b \mid c)(b \mid c))^*(b \mid c))  \mid  ((b \mid c)(b \mid c))^*(b \mid c)a((b \mid c)(b \mid c))^*$$

\section*{Exercise 34}
The set of strings over \{\textit{a,b,c}\} with an odd number of occurrences of the substring \textit{ab}.
\newline
$$((b \mid a^*c)^*a^*) ab ((b \mid a^*c)^*a^*) ( ab ((b \mid a^*c)^*a^*) ab ((b \mid a^*c)^*a^*))^*$$

\section*{Exercise 35}
The set of strings of odd length over \{\textit{a,b}\} that contain exactly two \textit{b}'s.
\newline
$$(aa)^*ab(aa)^*b(aa)^*  \mid  (aa)^*b(aa)^*ab(aa)  \mid  (aa)^*b(aa)^*b(aa)^*a  \mid  (aa)^*ab(aa)^*ab(aa)^*a$$

\section*{Exercise 36}
The set of strings over \{\textit{a,b}\} with an even number of \textit{a}'s or an odd number of \textit{b}'s.
\newline
$$(a^*ba^*)^*  \mid  (a^*ba^*)^*b(a^*ba^*)^*$$

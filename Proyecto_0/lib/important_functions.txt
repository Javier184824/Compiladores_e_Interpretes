;--------------------------------------------------------------------------------------------------
; Esta funcion se encarga de escribir texto en la consola
; Entradas: rdi -> Direccion de memoria del texto
;           rsi -> Cantidad de bytes a tomar del texto
; Salidas:  rax -> Retorna la cantidad de bytes usados
;               -> Si hubo un error retorna -1
write:
    ; Prologo
    push rbp            ; Guardar rbp en el stack
    mov rbp, rsp        ; Mover rsp a rbp

    ; Parte principal
    mov rdx, rsi        ; Cantidad de caracteres a escribir
    mov rsi, rdi        ; Direccion de memoria del texto
    mov rax, 1          ; 1 es el valor asignado para write
    mov rdi, 1          ; Usar stdout como file_descriptor
    syscall             ; Hacer la llamada a sistema write

    ; Fin
    pop rbp             ; Recuperar rbp del stack 
    ret                 ; Regresar al punto en donde se llamo la funcion
    
;--------------------------------------------------------------------------------------------------
; Esta funcion se encarga de recibir texto en la consola
; Entradas: rdi -> Direccion de memoria para guardar el texto
;           rsi -> Cantidad de bytes a tomar del texto
; Salidas:  rax -> Retorna la cantidad de bytes usados
;               -> Si hubo un error retorna -1
read:
    ; Prologo
    push rbp            ; Guardar rbp en el stack
    mov rbp, rsp        ; Mover rsp a rbp

    ; Parte principal
    mov rdx, rsi        ; Cantidad de caracteres a escribir
    mov rsi, rdi        ; Direccion de memoria en donde guardar el texto
    xor rax, rax        ; 0 es el valor asignado para read
    xor rdi, rdi        ; Usar stdin como file_descriptor
    syscall             ; Hacer la llamada a sistema write

    ; Fin
    pop rbp             ; Recuperar rbp del stack
    ret                 ; Regresar al punto en donde se llamo la funcion

; Entradas: rax -> Dirección de memoria a guardar el número
; Salidas:  rax -> Se guarda el número que se encontraba en la
;           dirección de memoria
atoi:
    push rbp
    push rbx                    ; Dirección de memoria
    push rcx                    ; Índice
    push r8                     ; Digito
    push r9                     ; Indicar si el número es negativo
    push r10                    ; Guarda el valor de 10
    mov rbp, rsp
    mov rbx, rax
    mov rax, 0
    mov rcx, 0
    mov r8, 0
    mov r9, 0
    mov r10, 10
    mov r8b, byte[rbx + rcx]
    cmp r8b, 2Dh
    je .isNegative
    jmp .addDigits
.isNegative:
    mov r9, 1
    inc rcx
.addDigits:
    mov r8b, byte[rbx + rcx]
    cmp r8b, 0Ah
    je .checkNegative
    sub r8b, 30h
    add rax, r8
    inc rcx
    mov r8b, byte[rbx + rcx]
    cmp r8b, 0Ah
    je .checkNegative
.multiplyTen:
    mul r10
    jmp .addDigits
.checkNegative:
    cmp r9, 1
    je .makeNegative
    jmp .end
.makeNegative:
    neg rax
.end:
    pop r10
    pop r9
    pop r8
    pop rcx
    pop rbx
    pop rbp
    ret
